    <script>
        // --- SYSTÈME DE PROGRESSION & XP ---
        let userStats = JSON.parse(localStorage.getItem('hggsp_stats')) || {
            xp: 0,
            level: 1,
            badges: { riddle: false, flash: false, time: false, actor: false }
        };

        const levels = [
            { min: 0, title: "Niveau 1 : Explorateur de la Wilderness" },
            { min: 100, title: "Niveau 2 : Conservateur Fédéral" },
            { min: 250, title: "Niveau 3 : Expert du Balancier Climatique" },
            { min: 500, title: "Niveau 4 : Maître HGGSP (Prêt pour le Bac)" }
        ];

        function addXP(amount, badgeKey = null, customTitle = "Action validée") {
            userStats.xp += amount;
            if (badgeKey && !userStats.badges[badgeKey]) {
                userStats.badges[badgeKey] = true;
                showToast("🏆 Badge Débloqué !", `Tu as remporté un nouveau trophée !`, 4000);
            } else {
                showToast(customTitle, `+${amount} XP gagnés !`);
            }

            // Calcul du niveau
            let currentLevelObj = levels[0];
            for (let i = 0; i < levels.length; i++) {
                if (userStats.xp >= levels[i].min) {
                    currentLevelObj = levels[i];
                    userStats.level = i + 1;
                }
            }

            localStorage.setItem('hggsp_stats', JSON.stringify(userStats));
            updateUI();
        }

        function updateUI() {
            const currentLevelObj = levels[userStats.level - 1];
            const nextLevelMin = levels[userStats.level] ? levels[userStats.level].min : currentLevelObj.min + 300;
            const currentLevelMin = currentLevelObj.min;
            const progressPercent = Math.min(100, Math.max(0, ((userStats.xp - currentLevelMin) / (nextLevelMin - currentLevelMin)) * 100));

            document.getElementById('level-title').innerText = currentLevelObj.title;
            document.getElementById('xp-text').innerText = `${userStats.xp} / ${nextLevelMin} XP`;
            document.getElementById('xp-bar').style.width = `${progressPercent}%`;

            const unlockedBadgesCount = Object.values(userStats.badges).filter(Boolean).length;
            document.getElementById('badge-count').innerText = `${unlockedBadgesCount}/4`;
        }

        function showToast(title, desc, duration = 3000) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-title').innerText = title;
            document.getElementById('toast-desc').innerText = desc;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0');
            }, duration);
        }

        function resetProgress() {
            if (confirm("Veux-tu vraiment réinitialiser toute ta progression ?")) {
                localStorage.removeItem('hggsp_stats');
                location.reload();
            }
        }

        // --- NAVIGATION TABS ---
        function switchTab(tabId) {
            document.querySelectorAll('main > section').forEach(sec => sec.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-emerald-600', 'text-white', 'shadow');
                btn.classList.add('bg-slate-700', 'text-slate-300');
            });
            document.getElementById(`section-${tabId}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`btn-${tabId}`);
            activeBtn.classList.remove('bg-slate-700', 'text-slate-300');
            activeBtn.classList.add('bg-emerald-600', 'text-white', 'shadow');
        }

        // --- BLOC 1 : ÉNIGMES ---
        const riddlesData = [
            { clue: "Je suis né en 1872 sous l'impulsion du président Grant, créant un précédent mondial pour sanctuariser la nature sans l'homme.", answer: "yellowstone" },
            { clue: "Je suis un accord international signé en 1987 sous Reagan pour protéger la couche d'ozone, considéré comme le plus efficace de l'histoire.", answer: "montreal" },
            { clue: "Je suis le concept géopolitique mené par la Californie et des coalitions d'États pour appliquer l'accord climatique malgré l'absence de Washington.", answer: "paradiplomatie" },
            { clue: "Je suis la révolution technologique de fracturation hydraulique (2009) qui a propulsé les USA au rang de superpuissance fossile.", answer: "schiste" }
        ];
        let currentRiddle = 0;

        function checkRiddle() {
            const userAns = document.getElementById('riddle-input').value.trim().toLowerCase();
            const feedback = document.getElementById('riddle-feedback');
            if (userAns.includes(riddlesData[currentRiddle].answer)) {
                feedback.className = "text-sm font-medium text-emerald-400 h-6";
                feedback.innerText = "Bravo ! Bonne réponse.";
                addXP(25, currentRiddle === riddlesData.length - 1 ? 'riddle' : null, "Énigme résolue !");
                setTimeout(() => {
                    currentRiddle = (currentRiddle + 1) % riddlesData.length;
                    document.getElementById('riddle-counter').innerText = `Énigme ${currentRiddle + 1} / 4`;
                    document.getElementById('riddle-clue').innerText = `"${riddlesData[currentRiddle].clue}"`;
                    document.getElementById('riddle-input').value = "";
                    feedback.innerText = "";
                }, 1500);
            } else {
                feedback.className = "text-sm font-medium text-rose-400 h-6";
                feedback.innerText = "Ce n'est pas tout à fait ça, réessaie !";
            }
        }

        // --- BLOC 2 : FLASHCARDS ---
        const flashcardsData = [
            { front: "Wilderness", back: "Espace sauvage perçu comme vierge, sacralisé par l'art (Bierstadt) et institutionnalisé par le Wilderness Act (1964). Revers : expulsion violente des Amérindiens." },
            { front: "Préservation vs Conservation", back: "Préservation (John Muir) : sanctuariser pour elle-même. Conservation (Roosevelt/Pinchot) : gérer rationnellement les ressources pour la puissance." },

        let currentCard = 0;
        let isFlipped = false;

        function flipCard() {
            isFlipped = !isFlipped;
            document.getElementById('flashcard-inner').style.transform = isFlipped ? 'rotateY(180deg)' : 'rotateY(0deg)';
            if (isFlipped) addXP(5, 'flash', "Flashcard consultée");
        }
        function updateCard() {
            isFlipped = false;
            document.getElementById('flashcard-inner').style.transform = 'rotateY(0deg)';
            document.getElementById('card-index').innerText = currentCard + 1;
            document.getElementById('card-front-text').innerText = flashcardsData[currentCard].front;
            document.getElementById('card-back-text').innerText = flashcardsData[currentCard].back;
        }
        function nextCard() { currentCard = (currentCard + 1) % flashcardsData.length; updateCard(); }
        function prevCard() { currentCard = (currentCard - 1 + flashcardsData.length) % flashcardsData.length; updateCard(); }

        // --- BLOC 3 : FRISE DU BALANCIER ---
        const timelineEvents = [
            { year: 1872, text: "Création de Yellowstone, premier parc national au monde." },
            { year: 1903, text: "Theodore Roosevelt et John Muir campent à Yosemite." },
            { year: 1964, text: "Vote du Wilderness Act par le Congrès." },
            { year: 1970, text: "Premier Earth Day et création de l'EPA par Nixon." },
            { year: 1987, text: "Protocole de Montréal (protection de la couche d'ozone)." },

            { year: 2026, text: "Sortie effective de l'Accord de Paris et retrait de la CCNUCC." }
        ];
        let shuffledEvents = [];

        function initTimeline() {
            shuffledEvents = [...timelineEvents].sort(() => Math.random() - 0.5);
            const container = document.getElementById('timeline-game');
            container.innerHTML = '';
            shuffledEvents.forEach((ev, idx) => {
                container.innerHTML += `
                    <div class="flex items-center gap-3 bg-slate-800 border border-slate-700 p-3 rounded-lg">
                        <select id="pos-${idx}" class="bg-slate-900 border border-slate-600 rounded p-1 text-sm text-white font-bold">
                            ${timelineEvents.map((_, i) => `<option value="${i+1}">#${i+1}</option>`).join('')}
                        </select>
                        <span class="text-sm text-slate-200">${ev.text}</span>
                    </div>
                `;
            });
        }

        function checkTimeline() {
            let correct = true;
            shuffledEvents.forEach((ev, idx) => {
                const selectedPos = parseInt(document.getElementById(`pos-${idx}`).value) - 1;
                const sortedByYear = [...timelineEvents].sort((a,b) => a.year - b.year);
                if (sortedByYear[selectedPos].year !== ev.year) correct = false;
            });
            const res = document.getElementById('timeline-result');
            if (correct) {
                res.className = "mt-3 text-sm font-medium text-emerald-400";
                res.innerText = "Parfait ! Chronologie validée.";
                addXP(30, 'time', "Frise chronologique validée !");
            } else {
                res.className = "mt-3 text-sm font-medium text-rose-400";
                res.innerText = "Il y a des erreurs dans l'ordre. Réessaie !";
            }
        }

        // --- BLOC 4 : JEU DES ACTEURS ---
        const actorChallenges = [
            { action: "Sortie effective de l'Accord de Paris le 27 janvier 2026 et retrait de la CCNUCC.", answer: "État fédéral (Trump II)", options: ["État fédéral (Trump II)", "U.S. Climate Alliance", "Firmes Transnationale (Exxon)", "ONG Greenpeace"] },
            { action: "Maintien de l'engagement climatique international via la U.S. Climate Alliance.", answer: "Paradiplomatie (États, Villes)", options: ["Le Congrès (Sénat)", "Paradiplomatie (États, Villes)", "ExxonMobil", "La Cour Suprême"] },
            { action: "Financement historique du climatoscepticisme (Global Climate Coalition) dès les années 1990.", answer: "Firmes Transnationales", options: ["État fédéral", "ONG Greenpeace", "Firmes Transnationales", "Nations Amérindiennes"] }
        ];
        let currentActorIdx = 0;

        function loadActorChallenge() {
            const current = actorChallenges[currentActorIdx];
            document.getElementById('actor-action-text').innerText = current.action;
            document.getElementById('actor-feedback').innerText = "";
            const choicesContainer = document.getElementById('actor-choices');
            choicesContainer.innerHTML = '';
            
            const shuffledOpts = [...current.options].sort(() => Math.random() - 0.5);
            shuffledOpts.forEach(opt => {
                choicesContainer.innerHTML += `
                    <button onclick="checkActorChoice('${opt.replace(/'/g, "\\'")}', '${current.answer.replace(/'/g, "\\'")}')" class="bg-slate-900 hover:bg-slate-700 border border-slate-700 p-3 rounded-lg text-sm text-slate-200 transition text-left font-medium">${opt}</button>
                `;
            });
        }

        function checkActorChoice(selected, correct) {
            const feedback = document.getElementById('actor-feedback');
            if (selected === correct) {
                feedback.className = "text-sm font-medium text-emerald-400 h-6";
                feedback.innerText = "Exact ! Bonne association.";
                addXP(15, currentActorIdx === actorChallenges.length - 1 ? 'actor' : null, "Défi Acteurs réussi !");
            } else {
                feedback.className = "text-sm font-medium text-rose-400 h-6";
                feedback.innerText = `Incorrect. C'était : ${correct}`;
            }
        }

        function nextActorChallenge() {
            currentActorIdx = (currentActorIdx + 1) % actorChallenges.length;
            loadActorChallenge();
        }

        // --- BLOC 5 : DISSERTATION ---
