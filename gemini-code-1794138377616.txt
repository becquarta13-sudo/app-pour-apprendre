    <script>
        // --- NAVIGATION TABS ---
        function switchTab(tabId) {
            document.querySelectorAll('main > section').forEach(sec => sec.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-emerald-600', 'text-white', 'shadow');
                btn.classList.add('bg-slate-700', 'text-slate-300');
            });
            document.getElementById(`section-${tabId}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`btn-${tabId}`);
            activeBtn.classList.remove('bg-slate-700', 'text-slate-300');
            activeBtn.classList.add('bg-emerald-600', 'text-white', 'shadow');
        }

        // --- MODULE 1 : FLASHCARDS ---
        const flashcardsData = [
            { front: "Wilderness", back: "Espace sauvage perçu comme vierge, sacralisé par l'art (Bierstadt) et institutionnalisé par le Wilderness Act (1964). Revers : expulsion violente des Amérindiens (William Cronon)." },
            { front: "Préservation vs Conservation", back: "Préservation (John Muir) : sanctuariser la nature pour elle-même. Conservation (Theodore Roosevelt / Gifford Pinchot) : gérer rationnellement les ressources pour la puissance économique." },
            { front: "Le Balancier Fédéral", back: "Alternance politique : l'État fédéral signe des accords sous les démocrates (Rio 1992, Paris 2015, Biden 2021) et s'en retire sous les républicains (Kyoto bloqué, Trump et la sortie de Paris en janv. 2026)." },
            { front: "La Révolution du Schiste", back: "Fracturation hydraulique (2009) propulsant les États-Unis au rang de superpuissance fossile (13,6 millions de barils/jour en 2025), utilisant le GNL comme levier géopolitique." },
            { front: "La Paradiplomatie", back: "Action internationale des contre-pouvoirs intérieurs (États fédérés et villes via la U.S. Climate Alliance, la Californie, Michael Bloomberg) qui maintiennent l'engagement malgré le retrait de Washington." },
            { front: "Le Transcendantalisme", back: "Mouvement philosophique (Emerson, Thoreau) des années 1830 voyant dans la nature un lien direct avec le divin et une régénération morale face à la civilisation." }
        ];
        let currentCard = 0;
        let isFlipped = false;

        function flipCard() {
            isFlipped = !isFlipped;
            const inner = document.getElementById('flashcard-inner');
            if (isFlipped) {
                inner.style.transform = 'rotateY(180deg)';
            } else {
                inner.style.transform = 'rotateY(0deg)';
            }
        }

        function updateCard() {
            isFlipped = false;
            document.getElementById('flashcard-inner').style.transform = 'rotateY(0deg)';
            document.getElementById('card-index').innerText = currentCard + 1;
            document.getElementById('card-front-text').innerText = flashcardsData[currentCard].front;
            document.getElementById('card-back-text').innerText = flashcardsData[currentCard].back;
        }

        function nextCard() {
            currentCard = (currentCard + 1) % flashcardsData.length;
            updateCard();
        }

        function prevCard() {
            currentCard = (currentCard - 1 + flashcardsData.length) % flashcardsData.length;
            updateCard();
        }

        // --- MODULE 2 : FRISE DU BALANCIER ---
        const timelineEvents = [
            { year: 1872, text: "Création de Yellowstone, premier parc national au monde." },
            { year: 1903, text: "Theodore Roosevelt et John Muir campent à Yosemite." },
            { year: 1964, text: "Vote du Wilderness Act par le Congrès." },
            { year: 1970, text: "Premier Earth Day et création de l'EPA par Nixon." },
            { year: 1987, text: "Protocole de Montréal (protection de la couche d'ozone)." },
            { year: 2015, text: "Signature de l'Accord de Paris sur le climat (Obama)." },
            { year: 2026, text: "Sortie effective de l'Accord de Paris et retrait de la CCNUCC (Trump II)." }
        ];
        let shuffledEvents = [];

        function initTimeline() {
            shuffledEvents = [...timelineEvents].sort(() => Math.random() - 0.5);
            const container = document.getElementById('timeline-game');
            container.innerHTML = '';
            shuffledEvents.forEach((ev, idx) => {
                container.innerHTML += `
                    <div class="flex items-center gap-3 bg-slate-800 border border-slate-700 p-3 rounded-lg">
                        <select id="pos-${idx}" class="bg-slate-900 border border-slate-600 rounded p-1 text-sm text-white font-bold">
                            ${timelineEvents.map((_, i) => `<option value="${i+1}">#${i+1}</option>`).join('')}
                        </select>
                        <span class="text-sm text-slate-200">${ev.text}</span>
                    </div>
                `;
            });
        }

        function checkTimeline() {
            let correct = true;
            shuffledEvents.forEach((ev, idx) => {
                const selectedPos = parseInt(document.getElementById(`pos-${idx}`).value) - 1;
                const correctYear = ev.year;
                const sortedByYear = [...timelineEvents].sort((a,b) => a.year - b.year);
                if (sortedByYear[selectedPos].year !== correctYear) {
                    correct = false;
                }
            });
            const res = document.getElementById('timeline-result');
            if (correct) {
                res.className = "mt-3 text-sm font-medium text-emerald-400";
                res.innerText = "Bravo ! La chronologie du balancier et des grandes lois est parfaitement restituée.";
            } else {
                res.className = "mt-3 text-sm font-medium text-rose-400";
                res.innerText = "Il y a des erreurs dans l'ordre chronologique. Réessayez !";
            }
        }

        // --- MODULE 3 : JEU DES ACTEURS ---
        const actorChallenges = [
            { action: "Sortie effective de l'Accord de Paris le 27 janvier 2026 et retrait de la CCNUCC.", answer: "État fédéral (Exécutif / Trump)", options: ["État fédéral (Exécutif / Trump)", "U.S. Climate Alliance", "Firmes Transnationales", "ONG The Nature Conservancy"] },
            { action: "Maintien de l'engagement climatique international et financement de la part américaine de la CCNUCC.", answer: "Paradiplomatie (États, Villes, Bloomberg)", options: ["Le Congrès (Sénat)", "Paradiplomatie (États, Villes, Bloomberg)", "ExxonMobil", "La Cour Suprême"] },
            { action: "Financement historique du climatoscepticisme (Global Climate Coalition) dès les années 1990.", answer: "Firmes Transnationales (Fossiles)", options: ["État fédéral", "ONG Greenpeace", "Firmes Transnationales (Fossiles)", "Nations Amérindiennes"] },
            { action: "Refus de ratifier le Protocole de Kyoto en adoptant la résolution Byrd-Hagel à 95-0.", answer: "Le Congrès (Sénat américain)", options: ["Le Président", "Le Congrès (Sénat américain)", "La Californie", "Le GIEC"] }
        ];
        let currentActorIdx = 0;

        function loadActorChallenge() {
            const current = actorChallenges[currentActorIdx];
            document.getElementById('actor-action-text').innerText = current.action;
            document.getElementById('actor-feedback').innerText = "";
            const choicesContainer = document.getElementById('actor-choices');
            choicesContainer.innerHTML = '';
            
            // Mélanger les options
            const shuffledOpts = [...current.options].sort(() => Math.random() - 0.5);
            shuffledOpts.forEach(opt => {
                choicesContainer.innerHTML += `
                    <button onclick="checkActorChoice('${opt.replace(/'/g, "\\'")}', '${current.answer.replace(/'/g, "\\'")}')" class="bg-slate-900 hover:bg-slate-700 border border-slate-700 p-3 rounded-lg text-sm text-slate-200 transition text-left font-medium">${opt}</button>
                `;
            });
        }

        function checkActorChoice(selected, correct) {
            const feedback = document.getElementById('actor-feedback');
            if (selected === correct) {
                feedback.className = "text-sm font-medium text-emerald-400 h-6";
                feedback.innerText = "Exact ! C'est le bon acteur pour cette dynamique.";
            } else {
                feedback.className = "text-sm font-medium text-rose-400 h-6";
                feedback.innerText = `Incorrect. La bonne réponse était : ${correct}`;
            }
        }

        function nextActorChallenge() {
            currentActorIdx = (currentActorIdx + 1) % actorChallenges.length;
            loadActorChallenge();
        }

        // --- MODULE 4 : DISSERTATION ---
        function revealCorrection() {
            document.getElementById('correction-box').classList.remove('hidden');
        }

        // Initialisation au chargement
        window.onload = () => {
            initTimeline();
            loadActorChallenge();
        };
    </script>
